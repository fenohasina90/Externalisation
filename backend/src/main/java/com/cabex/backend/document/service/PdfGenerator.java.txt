package com.cabex.backend.document.service;

import com.cabex.backend.demande.entity.DemandeService;
import com.cabex.backend.echeance.entity.EcheancePaiement;
import com.cabex.backend.echeance.entity.ProposPaiement;
import com.cabex.backend.paiement.entity.Paiement;
import com.lowagie.text.Document;
import com.lowagie.text.Element;
import com.lowagie.text.Font;
import com.lowagie.text.FontFactory;
import com.lowagie.text.PageSize;
import com.lowagie.text.Paragraph;

import java.io.ByteArrayOutputStream;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

@Component
public class PdfGenerator {

    private static final DateTimeFormatter DATE_FMT = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    private static final Font TITRE = FontFactory.getFont(FontFactory.HELVETICA_BOLD, 18);
    private static final Font SOUS_TITRE = FontFactory.getFont(FontFactory.HELVETICA, 11);
    private static final Font ENTETE = FontFactory.getFont(FontFactory.HELVETICA_BOLD, 11);
    private static final Font NORMAL = FontFactory.getFont(FontFactory.HELVETICA, 10);
    private static final Font BOLD = FontFactory.getFont(FontFactory.HELVETICA_BOLD, 10);

    public byte[] recapitulatifDemande(DemandeService d) {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Document doc = new Document(PageSize.A4);
        try {
            PdfWriter.getInstance(doc, out);
            doc.open();

            titre(doc, "Récapitulatif de demande");
            sousTitre(doc, "Référence : " + valeur(d.getReferenceDemande()));

            infoTable(doc, new String[][]{
                    {"Client", valeur(d.getSociete() != null ? d.getSociete().getNomSociete() : null)},
                    {"Adresse", valeur(d.getSociete() != null ? d.getSociete().getAdresse() : null)},
                    {"Contact client", valeur(d.getSociete() != null ? d.getSociete().getContact() : null)},
                    {"Responsable", valeur(d.getClientResponsable() != null ? d.getClientResponsable().getNomClient() : null)},
                    {"Commercial", valeur(d.getCommercial() != null ? d.getCommercial().getNomCommercial() : null)},
                    {"Date de demande", valeurDate(d.getDateDemande())},
                    {"Nombre de services", d.getLignes() != null ? String.valueOf(d.getLignes().size()) : "-"}
            });

            label(doc, "Services");
            PdfPTable table = new PdfPTable(4);
            table.setWidthPercentage(100);
            table.setWidths(new float[]{4f, 3f, 3f, 3f});
            table.setHeaderRows(1);
            enteteCellule(table, "Service");
            enteteCellule(table, "Période");
            enteteCellule(table, "Tarif");
            enteteCellule(table, "Statut");
            if (d.getLignes() != null) {
                for (com.cabex.backend.demande.entity.LigneService l : d.getLignes()) {
                    cellule(table, valeur(l.getService() != null ? l.getService().getNomService() : null));
                    cellule(table, valeurDate(l.getDebutService()) + " → " + valeurDate(l.getFinService()));
                    cellule(table, formatArgent(l.getTarif()));
                    cellule(table, valeur(l.getStatut()));
                }
            }
            doc.add(table);

            label(doc, "Informations de la demande");
            Paragraph p = new Paragraph(valeurNeOuVide(d.getCommentaire()), NORMAL);
            p.setSpacingAfter(10);
            doc.add(p);

            doc.close();
        } catch (Exception e) {
            throw new IllegalStateException("Erreur de génération du récapitulatif de demande", e);
        }
        return out.toByteArray();
    }

    public byte[] recuPaiement(Paiement p) {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Document doc = new Document(PageSize.A4);
        try {
            PdfWriter.getInstance(doc, out);
            doc.open();

            titre(doc, "Reçu de paiement");
            sousTitre(doc, "Référence paiement : " + valeur(p.getReferencePaiement()));

            infoTable(doc, new String[][]{
                    {"Client", valeur(p.getDemande() != null && p.getDemande().getSociete() != null
                            ? p.getDemande().getSociete().getNomSociete() : null)},
                    {"Demande", valeur(p.getDemande() != null ? p.getDemande().getReferenceDemande() : null)},
                    {"Montant", formatArgent(p.getMontantPaye())},
                    {"Date", valeurDate(p.getDatePaiement())},
                    {"Mode de paiement", valeur(p.getModePaiement() != null ? p.getModePaiement().getLibelle() : null)},
                    {"Référence virement", valeur(p.getReferenceVirement())},
                    {"Enregistré par", valeur(p.getCreatedBy() != null
                            ? p.getCreatedBy().getPrenom() + " " + p.getCreatedBy().getNom() : null)}
            });

            doc.close();
        } catch (Exception e) {
            throw new IllegalStateException("Erreur de génération du reçu de paiement", e);
        }
        return out.toByteArray();
    }

    public byte[] planPaiement(ProposPaiement propos) {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Document doc = new Document(PageSize.A4);
        try {
            PdfWriter.getInstance(doc, out);
            doc.open();

            titre(doc, "Plan de paiement");
            sousTitre(doc, "Demande : " + valeur(propos.getDemande() != null
                    ? propos.getDemande().getReferenceDemande() : null));

            infoTable(doc, new String[][]{
                    {"Montant total", formatArgent(propos.getMontantTotal())},
                    {"Reste à payer", formatArgent(propos.getRestePayer())},
                    {"Début", valeurDate(propos.getDateDebut())},
                    {"Fin", valeurDate(propos.getDateFin())},
                    {"Nombre d'échéances", propos.getNombreEcheances() != null
                            ? propos.getNombreEcheances().toString() : "-"}
            });

            label(doc, "Échéances");
            PdfPTable table = new PdfPTable(4);
            table.setWidthPercentage(100);
            table.setWidths(new float[]{3f, 3f, 3f, 3f});
            table.setHeaderRows(1);
            enteteCellule(table, "N°");
            enteteCellule(table, "Date");
            enteteCellule(table, "Montant attendu");
            enteteCellule(table, "Montant payé");
            if (propos.getEcheances() != null) {
                for (EcheancePaiement e : propos.getEcheances()) {
                    cellule(table, e.getNumeroEcheance() != null ? e.getNumeroEcheance().toString() : "-");
                    cellule(table, e.getDateEcheance() != null ? e.getDateEcheance().format(DATE_FMT) : "-");
                    cellule(table, formatArgent(e.getMontantAttendu()));
                    cellule(table, formatArgent(e.getMontantPaye()));
                }
            }
            doc.add(table);

            doc.close();
        } catch (Exception e) {
            throw new IllegalStateException("Erreur de génération du plan de paiement", e);
        }
        return out.toByteArray();
    }

    private static final Font PETIT = FontFactory.getFont(FontFactory.HELVETICA, 7);
    private static final Font PETIT2 = FontFactory.getFont(FontFactory.HELVETICA, 5);
    private static final Font PETIT_GRAS = FontFactory.getFont(FontFactory.HELVETICA_BOLD, 7);
    private static final Font PETIT_ITALIQUE =
            FontFactory.getFont(FontFactory.HELVETICA_OBLIQUE, 8);
    private static final Font PETIT_GRAS_ITALIQUE =
            FontFactory.getFont(FontFactory.HELVETICA_BOLDOBLIQUE, 8);

    private final com.cabex.backend.configuration.service.ConfigurationService configurationService;

    public PdfGenerator(com.cabex.backend.configuration.service.ConfigurationService configurationService) {
        this.configurationService = configurationService;
    }

    /**
     * Facture proforma au format du modèle CABEX : en-tête (logo, date,
     * facture, destinataire), tableau des prestations, totaux HT/TVA/TTC,
     * montant en lettres, observations, conditions de règlement et signatures.
     */
    public byte[] factureProforma(com.cabex.backend.facture.entity.Facture f) {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Document doc = new Document(PageSize.A4, 28, 28, 24, 24);
        try {
            PdfWriter.getInstance(doc, out);
            doc.open();

            com.cabex.backend.demande.entity.DemandeService demande = f.getDemande();
            com.cabex.backend.client.entity.Societe societe =
                    demande != null ? demande.getSociete() : null;

            // ---- En-tête : logo | date+facture | destinataire ----
            PdfPTable entete = new PdfPTable(3);
            entete.setWidthPercentage(100);
            entete.setWidths(new float[]{2.2f, 2.4f, 3.4f});
            entete.addCell(celluleLogo());
            entete.addCell(celluleSansBordure(new Paragraph(
                    "DATE : " + valeurDate(f.getCreatedAt() != null
                            ? f.getCreatedAt().toLocalDate() : null)
                            + "\nFACTURE :\n" + valeur(f.getReferenceFacture()), PETIT_GRAS)));
            StringBuilder doit = new StringBuilder("DOIT:\nRAISON SOCIAL: "
                    + (societe != null ? societe.getNomSociete() : "-"));
            if (societe != null) {
                doit.append("\nADRESSE: ").append(valeurTexte(societe.getAdresse()));
                doit.append("\nCONTACT: ").append(valeurTexte(societe.getContact()));
                doit.append("\nMAIL: ").append(valeurTexte(societe.getEmail()));
            }
            doit.append("\nNIF :\nSTATISTIQUE :\nRCS :");
            if (demande != null && demande.getClientResponsable() != null) {
                doit.append("\nResp: ").append(demande.getClientResponsable().getNomClient());
            }
            entete.addCell(celluleSansBordure(new Paragraph(doit.toString(), PETIT)));
            doc.add(entete);

            Paragraph activite = new Paragraph("ETUDES ET EXTERNALISATION DE SERVICES", PETIT_GRAS);
            activite.setSpacingBefore(4);
            activite.setSpacingAfter(6);
            doc.add(activite);

            // ---- Tableau des prestations ----
            PdfPTable table = new PdfPTable(6);
            table.setWidthPercentage(100);
            table.setWidths(new float[]{4.4f, 1.5f, 1.1f, 1.8f, 1.4f, 1.8f});
            table.setHeaderRows(1);
            enteteCellule(table, "DESIGNATION");
            enteteCellule(table, "PU (Ariary)");
            enteteCellule(table, "Quantité");
            enteteCellule(table, "Montant HTVA (Ariary)");
            enteteCellule(table, "Remise (Ariary)");
            enteteCellule(table, "Net commercial (Ariary)");

            celluleEtendue(table, "Prestation:", 6, true);
            java.math.BigDecimal totalHt = java.math.BigDecimal.ZERO;
            if (f.getLignes() != null) {
                for (com.cabex.backend.demande.entity.LigneService l : f.getLignes()) {
                    String designation = valeur(l.getService() != null
                            ? l.getService().getNomService() : null)
                            + "\n" + valeurDate(l.getDebutService())
                            + " → " + valeurDate(l.getFinService());
                    cellule(table, designation);
                    cellule(table, formatMontant(l.getTarif()));
                    cellule(table, "1");
                    cellule(table, formatMontant(l.getTarif()));
                    cellule(table, "-");
                    cellule(table, formatMontant(l.getTarif()));
                    totalHt = totalHt.add(montantOuZero(l.getTarif()));
                }
            }
            ligneSousTotal(table, "Sous-total Prestation", totalHt);

            java.math.BigDecimal tauxTva = lireTauxTva();
            java.math.BigDecimal tva = totalHt.multiply(tauxTva)
                    .divide(new java.math.BigDecimal("100"), 2, RoundingMode.HALF_UP);
            java.math.BigDecimal ttc = totalHt.add(tva);
            java.math.BigDecimal avance = avanceRecue(f);
            java.math.BigDecimal netAPayer = ttc.subtract(avance).max(java.math.BigDecimal.ZERO);

            ligneTotal(table, "TOTAL HT", totalHt);
            ligneTotal(table, "TVA" + (tauxTva.signum() > 0 ? " (" + tauxTva.stripTrailingZeros()
                    .toPlainString() + " %)" : ""), tva);
            ligneTotal(table, "TOTAL TTC", ttc);
            ligneTotal(table, "Avance reçue", avance);
            ligneTotal(table, "Net à payer", netAPayer);
            doc.add(table);

            Paragraph arrete = new Paragraph(
                    "Arrêtée la présente facture à la somme de "
                            + MontantEnLettres.convertir(netAPayer.longValue()).toUpperCase()
                            + " ARIARY",
                    PETIT);
            arrete.setSpacingBefore(6);
            arrete.setSpacingAfter(4);
            doc.add(arrete);

            Paragraph observations = new Paragraph("OBSERVATIONS:\n"
                    + (demande != null ? valeurTexte(demande.getCommentaire()) : "-"), PETIT);
            observations.setSpacingAfter(2);
            doc.add(observations);

            // Espace réservé à l'écriture manuscrite sous les observations.
            Paragraph espaceObservations = new Paragraph(" ", PETIT);
            espaceObservations.setSpacingBefore(26);
            espaceObservations.setSpacingAfter(2);
            doc.add(espaceObservations);

            // ---- Conditions de règlement ----
            PdfPTable conditions = new PdfPTable(2);
            conditions.setWidthPercentage(100);
            conditions.setWidths(new float[]{1f, 1.4f});
            conditions.addCell(new PdfPCell(new Paragraph(
                    "Condition de règlement :\n"
                            + "Compte: BAOBAB BANQUE MADAGASCAR Ambodivona\n"
                            + "00013 - 00011 - 24798687313 - 64\n"
                            + "Au nom de : CABINET EXCELLENT\n"
                            + "LOT LI O 154 J C AVARADOHA ANTANANARIVO 101",
                    PETIT)));
            PdfPCell modes = new PdfPCell();
            modes.addElement(new Paragraph(
                    "□ par chèque        Nom Banque .......    N° CHQ ..........    Date ..............    Montant ............\n"
                            + "□ par virement        Nom Banque .......    Date ..............    Montant ............\n"
                            + "□ par espèce    Au nom de : CABINET EXCELLENT\n"
                            + "□ par mobile money",
                    PETIT));
            Paragraph echeance = new Paragraph("A payer dès la réception de la facture",
                    PETIT_GRAS_ITALIQUE);
            echeance.setAlignment(Element.ALIGN_CENTER);
            echeance.setSpacingBefore(4);
            modes.addElement(echeance);
            conditions.addCell(modes);
            doc.add(conditions);

            // ---- Bas de page : groupe + signatures + mentions (modèle CABEX) ----
            PdfPTable bas = new PdfPTable(5);
            bas.setWidthPercentage(100);
            bas.setWidths(new float[]{1.7f, 0.9f, 1.3f, 1.3f, 0.9f});

            Paragraph groupe = new Paragraph();
            groupe.add(new Phrase("Société Groupe Excellent :\n", PETIT_GRAS));
            groupe.add(new Phrase("Capital : 222 126 000 Ar\n", PETIT_ITALIQUE));
            groupe.add(new Phrase(
                    "Agence Immo Excellent\n"
                            + "Agence de recrutement Excellent\n"
                            + "Agence Fiduciaire Excellent\n"
                            + "Institut Supérieur Excellent\n"
                            + "Tantsaha Production\n"
                            + "Cabinet Nakay Mahajanga\n"
                            + "Restaurant Nakay Mahajanga\n"
                            + "Association Nakay",
                    PETIT_GRAS));
            PdfPCell groupeCell = new PdfPCell(groupe);
            groupeCell.setRowspan(3);
            bas.addCell(groupeCell);

            bas.addCell(centrer("Client", PETIT_GRAS));
            bas.addCell(centrer("Responsable Traitement\nDossiers Clients", PETIT_GRAS));
            bas.addCell(centrer("Responsable Audit\nContrôleur de Gestion", PETIT_GRAS));
            bas.addCell(centrer("Manager", PETIT_GRAS));

            // Espace réservé aux signatures manuscrites.
            PdfPCell espaceSignatures = new PdfPCell(new Paragraph(" ", PETIT));
            espaceSignatures.setColspan(4);
            espaceSignatures.setFixedHeight(52);
            bas.addCell(espaceSignatures);
            

            Paragraph mentions = new Paragraph();
            mentions.setAlignment(Element.ALIGN_CENTER);
            mentions.add(new Phrase(
                    "NIF : 6001199978   STAT : 71102 11 2013 0 02016   RCS : 2014A01518\n"
                            + "Adresse siège : LOT II P 154 JC AVARADOHA\n"
                            + "Antananarivo 101, Madagascar\n",
                    PETIT));
            mentions.add(new Phrase("Mail: accueil.cabex@gmail.com\n", PETIT_GRAS));
            mentions.add(new Phrase("facebook : Cabinet Excellent", PETIT));
            PdfPCell mentionsCell = new PdfPCell(mentions);
            mentionsCell.setColspan(3);
            bas.addCell(mentionsCell);

            Paragraph contacts = new Paragraph(
                    "Contact Tanà : 034 44 400 22\n"
                            + "Contact Majunga : 034 32 811 66\n"
                            + "Mail : cabinet.nakay@gmail.com",
                    PETIT2);
            bas.addCell(contacts);
            doc.add(bas);



            doc.close();
        } catch (Exception e) {
            throw new IllegalStateException("Erreur de génération de la facture proforma", e);
        }
        return out.toByteArray();
    }

    private void celluleEtendue(PdfPTable table, String texte, int portee, boolean gras) {
        PdfPCell c = new PdfPCell(new Paragraph(texte, gras ? PETIT_GRAS : PETIT));
        c.setColspan(portee);
        c.setPadding(4);
        table.addCell(c);
    }

    private void ligneSousTotal(PdfPTable table, String libelle, java.math.BigDecimal montant) {
        PdfPCell lib = new PdfPCell(new Paragraph(libelle, PETIT_GRAS));
        lib.setColspan(3);
        lib.setHorizontalAlignment(Element.ALIGN_RIGHT);
        lib.setPadding(4);
        table.addCell(lib);
        cellule(table, formatMontant(montant));
        cellule(table, "-");
        cellule(table, formatMontant(montant));
    }

    private void ligneTotal(PdfPTable table, String libelle, java.math.BigDecimal montant) {
        PdfPCell lib = new PdfPCell(new Paragraph(libelle, PETIT_GRAS));
        lib.setColspan(3);
        lib.setPadding(4);
        table.addCell(lib);
        cellule(table, "");
        cellule(table, "");
        cellule(table, formatMontant(montant));
    }

    private PdfPCell centrer(String texte, Font police) {
        Paragraph p = new Paragraph(texte, police);
        p.setAlignment(Element.ALIGN_CENTER);
        return new PdfPCell(p);
    }

    private PdfPCell celluleSansBordure(Paragraph contenu) {
        PdfPCell c = new PdfPCell(contenu);
        c.setBorder(Rectangle.NO_BORDER);
        c.setPadding(2);
        return c;
    }

    private PdfPCell celluleLogo() {
        try (java.io.InputStream in =
                     getClass().getResourceAsStream("/static/logo-cabinet-excellent.jpg")) {
            if (in != null) {
                com.lowagie.text.Image image =
                        com.lowagie.text.Image.getInstance(in.readAllBytes());
                image.scaleToFit(85, 85);
                PdfPCell c = new PdfPCell(image, true);
                c.setBorder(Rectangle.NO_BORDER);
                c.setPadding(2);
                return c;
            }
        } catch (Exception e) {
            // Repli : nom du cabinet en texte.
        }
        return celluleSansBordure(new Paragraph("CABINET EXCELLENT", PETIT_GRAS));
    }

    private String valeurTexte(String s) {
        return s == null || s.isBlank() ? "" : s;
    }

    private String formatMontant(java.math.BigDecimal montant) {
        if (montant == null) {
            return "-";
        }
        return montant.setScale(0, RoundingMode.HALF_UP).toPlainString();
    }

    private java.math.BigDecimal montantOuZero(java.math.BigDecimal montant) {
        return montant != null ? montant : java.math.BigDecimal.ZERO;
    }

    private java.math.BigDecimal lireTauxTva() {
        try {
            String valeur = configurationService.lireValeur("TAUX_TVA", "0");
            return new java.math.BigDecimal(valeur.trim()).max(java.math.BigDecimal.ZERO);
        } catch (Exception e) {
            return java.math.BigDecimal.ZERO;
        }
    }

    /** Avance déjà reçue : somme payée sur les échéances des lignes facturées. */
    private java.math.BigDecimal avanceRecue(com.cabex.backend.facture.entity.Facture f) {
        java.math.BigDecimal avance = java.math.BigDecimal.ZERO;
        if (f.getLignes() == null) {
            return avance;
        }
        for (com.cabex.backend.demande.entity.LigneService l : f.getLignes()) {
            if (l.getPlansPaiement() == null) {
                continue;
            }
            for (com.cabex.backend.echeance.entity.ProposPaiement plan : l.getPlansPaiement()) {
                if (plan.getEcheances() == null) {
                    continue;
                }
                for (com.cabex.backend.echeance.entity.EcheancePaiement e : plan.getEcheances()) {
                    avance = avance.add(montantOuZero(e.getMontantPaye()));
                }
            }
        }
        return avance;
    }

    private void titre(Document doc, String text) {
        Paragraph p = new Paragraph(text, TITRE);
        p.setAlignment(Element.ALIGN_CENTER);
        p.setSpacingAfter(4);
        doc.add(p);
    }

    private void sousTitre(Document doc, String text) {
        Paragraph p = new Paragraph(text, SOUS_TITRE);
        p.setSpacingAfter(14);
        doc.add(p);
    }

    private void label(Document doc, String text) {
        Paragraph p = new Paragraph(text, BOLD);
        p.setSpacingBefore(12);
        p.setSpacingAfter(6);
        doc.add(p);
    }

    private void infoTable(Document doc, String[][] lignes) {
        PdfPTable table = new PdfPTable(2);
        table.setWidthPercentage(100);
        table.setWidths(new float[]{3f, 7f});
        for (String[] ligne : lignes) {
            PdfPCell k = new PdfPCell(new Phrase(ligne[0], BOLD));
            k.setPadding(6);
            k.setBackgroundColor(new java.awt.Color(238, 238, 238));
            table.addCell(k);
            table.addCell(new PdfPCell(new Phrase(ligne[1], NORMAL)));
        }
        doc.add(table);
    }

    private void enteteCellule(PdfPTable table, String text) {
        PdfPCell c = new PdfPCell(new Phrase(text, ENTETE));
        c.setBackgroundColor(new java.awt.Color(220, 220, 220));
        c.setPadding(6);
        table.addCell(c);
    }

    private void cellule(PdfPTable table, String text) {
        PdfPCell c = new PdfPCell(new Phrase(text, NORMAL));
        c.setPadding(6);
        table.addCell(c);
    }

    private String formatArgent(BigDecimal montant) {
        if (montant == null) {
            return "-";
        }
        return montant.setScale(2, RoundingMode.HALF_UP).toString() + " (MGA)";
    }

    private String valeur(String s) {
        return s == null || s.isBlank() ? "-" : s;
    }

    private String valeurDate(LocalDate date) {
        return date == null ? "-" : date.format(DATE_FMT);
    }

    private String valeurNeOuVide(String s) {
        return s == null || s.isBlank() ? "-" : s;
    }
}
